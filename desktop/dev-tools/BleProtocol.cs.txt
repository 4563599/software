        public static Frame Decode(string hex)
        {
            var frame = new Frame();
            string text = (hex ?? "").Trim();
            frame.IsRedacted = Regex.IsMatch(text, "redact|脱敏|已隐藏|\\*\\*|••", RegexOptions.IgnoreCase);
            if (frame.IsRedacted)
            {
                frame.Function = "报文内容已脱敏";
                frame.Error = "日志未提供完整报文字节，无法校验或逐字段解析。";
                return frame;
            }
            text = Regex.Replace(Regex.Replace(text, "0x", "", RegexOptions.IgnoreCase), "[\\s:,_-]", "");
            string prefix = Regex.Match(text, "^[0-9A-Fa-f]*").Value;
            if (prefix.Length > 2048)
            {
                frame.Error = "报文超过协议上限 1024 字节。";
                return frame;
            }
            var b = new byte[prefix.Length / 2];
            for (int i = 0; i < b.Length; i++) b[i] = Convert.ToByte(prefix.Substring(i * 2, 2), 16);
            frame.Bytes = b;
            if (b.Length < 2 || b[0] != 0xFB)
            {
                frame.Error = b.Length == 0 ? "没有可读取的十六进制字节。" : "未找到完整的 0xFB 明文业务帧头。";
                return frame;
            }
            if (b[1] != 0x03 && b[1] != 0x06)
            {
                frame.Function = "未覆盖的帧版本";
                frame.Error = "当前解码覆盖 FB/03 和 FB/06 明文帧。";
                return frame;
            }
            bool v6 = b[1] == 0x06;
            int page = v6 ? 2 : 0;
            frame.Pages = v6 ? new[] { 2, 3 } : new int[0];
            Explain(Add(frame, 0, 1, "帧头", "0xFB", "明文业务帧标识。", page, true),
                "位置 0 的字节 0xFB 是每一帧固定的第一个字节，看到它就知道一帧从这里开始。");
            Explain(Add(frame, 1, 1, "帧版本", Hex(b[1]), "SDK 支持的明文业务帧版本。", page, true),
                "位置 1 的字节 " + Hex(b[1]) + " = 十进制 " + b[1] + " → 第 " + b[1] + " 版协议（V0" + b[1] + "）。");
            if (b.Length > 2)
            {
                frame.Sequence = b[2];
                Explain(Add(frame, 2, 1, "序列号", Hex(b[2]) + "（" + b[2] + "）", "用于同一连接内匹配请求和应答。", page, true),
                    "位置 2 的字节 " + Hex(b[2]) + " = 十进制 " + b[2] + "。", "APP 每发一条加 1；设备应答时带回同一个序列号，所以请求和应答靠它配对。");
            }
            if (b.Length > 3) Explain(Add(frame, 3, 1, "Mode", Hex(b[3]), "V06 文档固定为 0。", page, true), "位置 3 的字节在 V06 协议里固定是 0x00。");
            if (b.Length > 4)
            {
                frame.Command = b[4];
                bool request = b[4] >= 0xA0 && b[4] <= 0xBF;
                Explain(Add(frame, 4, 1, "指令", Hex(b[4]), "外层业务指令码。", page, true),
                    "位置 4 的字节 " + Hex(b[4]) + " = 十进制 " + b[4] + "，决定后面数据域怎么解析。",
                    request ? "0xA0–0xBF 是 APP 发给设备的请求；设备的应答指令 = " + Hex(b[4]) + " − 0xA0 = " + Hex(b[4] - 0xA0) + "。"
                        : "小于 0xA0 是设备发给 APP 的；对应的请求指令 = " + Hex(b[4]) + " + 0xA0 = " + Hex(b[4] + 0xA0) + "。");
            }
            if (b.Length < 7)
            {
                frame.Error = "帧头不完整，无法读取业务数据长度。";
                return frame;
            }
            int length = (int)ReadNumber(b, 5, 2);
            Explain(Add(frame, 5, 2, "数据域长度", length + " 字节", "两字节小端。总帧长度为数据域长度加 9。", page, true),
                NumberSteps(b, 5, 2, length).Concat(new[] { "数据域 " + length + " 字节；整帧 = 帧头 7 字节 + 数据域 " + length + " + CRC 2 字节 = " + (length + 9) + " 字节。" }).ToArray());
            if (prefix.Length != text.Length || text.Length % 2 != 0 || b.Length != length + 9)
            {
                frame.Error = "帧不完整或长度不匹配。预期 " + (length + 9) + " 字节，读取到 " + b.Length + " 个完整字节。";
                return frame;
            }
            int crcOffset = b.Length - 2;
            int crc = 0;
            for (int i = 0; i < crcOffset; i++)
            {
                crc ^= b[i];
                for (int bit = 0; bit < 8; bit++) crc = (crc & 1) != 0 ? (crc >> 1) ^ 0xA001 : crc >> 1;
            }
            frame.ChecksumOk = ReadNumber(b, crcOffset, 2) == (ulong)crc;
            if (!frame.ChecksumOk.Value) frame.Error = "CRC16 校验不通过，业务字段仅供定位原始字节。";
            if (v6) DecodePayload(frame, b[4], length);
            else
            {
                frame.Function = "V03 业务字段未覆盖";
                Raw(frame, 7, length, 0);
            }
            int stored = (int)ReadNumber(b, crcOffset, 2);
            Explain(Add(frame, crcOffset, 2, "CRC16", frame.ChecksumOk.Value ? "通过" : "不通过", "覆盖帧头至数据域末尾，初始值 0，反射多项式 0xA001。计算值（小端） " + (crc & 255).ToString("X2") + " " + (crc >> 8).ToString("X2") + "。", v6 ? 3 : 0, true),
                "帧里最后两个字节（位置 " + crcOffset + " 的 " + b[crcOffset].ToString("X2") + "、位置 " + (crcOffset + 1) + " 的 " + b[crcOffset + 1].ToString("X2") + "）是校验码，按小端存储：从位置 " + (crcOffset + 1) + " 往回读到位置 " + crcOffset + "，拼成 0x" + stored.ToString("X4") + "。",
                "把位置 0–" + (crcOffset - 1) + " 的字节按 CRC16（初始值 0，多项式 0xA001）重新算一遍，得到 0x" + crc.ToString("X4") + "。",
                stored == crc ? "两个值相同 → 传输过程中没有出错，校验通过。" : "两个值不同 → 这一帧在传输或记录时出了错，后面的解析只能参考。");
            frame.Steps = "1. 位置 0 是帧头 0xFB，位置 1 是版本 " + Hex(b[1]) + "，位置 2 是序列号 " + Hex(b[2]) + "，位置 4 是指令 " + Hex(b[4]) + "。\n"
                + "2. 位置 5–6 的 " + b[5].ToString("X2") + " " + b[6].ToString("X2") + " 是数据域长度（小端，从位置 6 往回读）= " + length + " 字节；整帧 = 7 + " + length + " + 2 = " + b.Length + " 字节。\n"
                + "3. 用位置 0–" + (crcOffset - 1) + " 算出 CRC16，和最后两个字节（位置 " + crcOffset + "–" + (crcOffset + 1) + "）比较，结果" + (frame.ChecksumOk.Value ? "通过。" : "不通过。") + "\n4. "
                + (frame.Steps.Length > 0 ? frame.Steps : "根据位置 4 的指令 " + Hex(b[4]) + " 解析位置 7 开始的数据域。");
            return frame;
        }

        private static void DecodePayload(Frame f, int cmd, int n)
        {
            switch (cmd)
            {
                case 0xAA: case 0x0A: DecodeParameter(f, cmd == 0x0A, n); break;
                case 0xB6: case 0x16: DecodeFaults(f, cmd == 0x16, n); break;
                case 0xB7: case 0x17: DecodeRcd(f, cmd == 0x17, n); break;
                case 0xA8: case 0x08: DecodeRealtime(f, cmd == 0x08, n); break;
                default: f.Function = "未覆盖的指令 " + Hex(cmd); Raw(f, 7, n, 0); break;
            }
        }

        private static void DecodeParameter(Frame f, bool reply, int n)
        {
            int page = reply ? 14 : 9;
            f.Pages = new[] { 9, 13, 14, 2, 3 };
            f.Function = reply ? "桩信息设置/查询应答" : "桩信息设置/查询";
            if (n < 4) { Invalid(f, n, page, "参数数据域至少需要 4 字节。"); return; }
            byte[] b = f.Bytes;
            int control = b[7], address = b[8], gun = control & (reply ? 3 : 7), count = control >> 4;
            bool set = (control & (reply ? 4 : 8)) != 0;
            bool success = !reply || (control & 8) == 0;
            int m = (int)ReadNumber(b, 9, 2);
            string action = set ? "设置" : "查询";
            f.Address = address;
            f.Arguments = "枪号 " + gun + " · " + action + " · " + count + " 项 · 地址 " + Hex(address) + "（" + address + "）";
            // One byte holds several small values; each is read from its own bits.
            var controlSteps = new List<string> { ByteAt(7, control) + "。这 8 个 0/1 从右往左依次叫 bit0 到 bit7，不同的几位表示不同的内容：" };
            if (reply)
            {
                controlSteps.Add(BitPart(control, 0, 2) + "，就是 " + gun + " → 枪号 " + gun + "。");
                controlSteps.Add(BitPart(control, 2, 1) + " → 0 是查询、1 是设置 → " + action + "。");
                controlSteps.Add(BitPart(control, 3, 1) + " → 0 是成功、1 是失败 → " + (success ? "成功" : "失败") + "。");
            }
            else
            {
                controlSteps.Add(BitPart(control, 0, 3) + "，就是 " + gun + " → 枪号 " + gun + "。");
                controlSteps.Add(BitPart(control, 3, 1) + " → 0 是查询、1 是设置 → " + action + "。");
            }
            controlSteps.Add(BitPart(control, 4, 4) + "，就是 " + count + " → 这条消息带 " + count + " 个参数。");
            Explain(Add(f, 7, 1, "枪号 / 类型 / " + (reply ? "结果 / " : "") + "参数个数", "枪号 " + gun + " · " + action + (reply ? (success ? " · 成功" : " · 失败") : "") + " · " + count + " 项",
                reply ? "掩码 0x03 枪号，0x04 类型，0x08 结果，0xF0 参数个数。" : "掩码 0x07 枪号，0x08 类型，0xF0 参数个数。", page), controlSteps.ToArray());
            Describe(f, address, set);
            string parameter = f.Function.Length > 2 ? f.Function.Substring(2) : f.Function;
            Explain(Add(f, 8, 1, "参数地址", Hex(address) + "（" + address + "）", "在 §2.9 参数地址列表中匹配。", page),
                Hex(address) + " = 十进制 " + address + "。", "在 §2.9 参数地址表里查 " + address + " → 「" + parameter + "」。");
            Explain(Add(f, 9, 2, "参数内容长度", m + " 字节", address == 0xBF ? "两字节小端。键值查询可携带查询键。" : "两字节小端。普通查询请求长度为 0。", page),
                NumberSteps(b, 9, 2, m).Concat(new[] { m == 0 ? "长度为 0 → 这是查询请求，后面没有参数内容。" : "后面紧跟 " + m + " 字节参数内容（从位置 11 开始）。" }).ToArray());
            f.Steps = (reply ? "按 §2.10 拆解位置 7 的应答控制字节" : "按 §2.9 拆解位置 7 的请求控制字节") + "，位置 8 的参数地址是 " + Hex(address) + "。";
            if (count != 1 || m != n - 4)
            {
                Error(f, count == 1 ? "参数内容长度与实际字节数不一致。" : "当前字段表支持单参数消息。");
                Raw(f, 11, n - 4, page);
                return;
            }
            string value = m > 0 ? ParameterValue(f, address, m, page, set) : "";
            if (reply) f.Result = success ? action + "成功" + (value.Length > 0 ? " · " + value : "") : action + "失败";
            else if (value.Length > 0) f.Result = "参数值 · " + value;
        }

        private static void Describe(Frame f, int address, bool set)
        {
            string name, method = "", decoder = "";
            int[] doc;
            switch (address)
            {
                case 0x01: name = "SN 号"; method = set ? "" : "getDeviceId"; decoder = "ENBLEChargeInfoUnPack.parseDeviceId"; doc = new[] { 9 }; break;
                case 0x02: name = "当前时间"; method = set ? "setV6Time" : "getV6Time"; decoder = "ENBLEChargeInfoUnPack." + (set ? "parseCommonSetResponse" : "parseGetV6DeviceTimeResponse"); doc = new[] { 10 }; break;
                case 0x0B: name = "最大限制电流"; method = (set ? "set" : "get") + "MaxCurrent"; decoder = "ENBLEChargeInfoUnPack.parseGetDeviceMaxCurrentResponse"; doc = new[] { 10 }; break;
                case 0x0D: name = "桩模式"; method = (set ? "set" : "get") + "Mode"; decoder = "ENBLEChargeInfoUnPack.parseGetDeviceModeResponse"; doc = new[] { 10 }; break;
                case 0x1F: name = "ATE 模式操作"; f.Api = "ENBLESystemService.setATEMode"; decoder = "ENBLESystemUnPack.parseCommonSetResponse"; doc = new[] { 11 }; break;
                case 0x42: name = "负载平衡 / 光伏参数"; method = set ? "" : "getLoadBalance"; decoder = "ENBLEChargeInfoUnPack.parseGetLoadBalanceResponse"; doc = new[] { 11 }; break;
                case 0x4B: name = "电网类型"; method = (set ? "set" : "get") + "GridType"; decoder = "ENBLEChargeInfoUnPack.parseGetGridTypeResponse"; doc = new[] { 11 }; break;
                case 0x53: name = "过压阈值"; method = (set ? "set" : "get") + "OverVoltageThreshold"; decoder = "ENBLEChargeInfoUnPack.parseGetOverVoltageThresholdResponse"; doc = new[] { 11 }; break;
                case 0xBF: name = "键值参数"; doc = new[] { 12, 22, 23 }; break;
                case 0xC3: name = "状态"; f.Api = "SdkInstallationDetailsBleClient.getConnectedStatus"; decoder = "ENBLEChargeNetworkUnPack.parseGetConnectedStatusResponse"; doc = new[] { 12, 13, 19, 20 }; break;
                case 0xC5: name = "设备基础信息"; method = set ? "" : "getDeviceBasicInfo"; decoder = "ENBLEChargeInfoUnPack.parseGetDeviceBasicInfoResponse"; doc = new[] { 13 }; break;
                default: name = "参数 " + Hex(address) + "（字段未覆盖）"; doc = new int[0]; break;
            }
            if (method.Length > 0) f.Api = "ENBLEChargeInfoService." + method;
            if (!set || address == 0x02 || address == 0x1F) f.Decoder = decoder;
            f.Function = (set ? "设置" : "查询") + name;
            var pages = new List<int>(doc);
            foreach (int page in f.Pages) if (!pages.Contains(page)) pages.Add(page);
            f.Pages = pages.ToArray();
        }

        private static string ParameterValue(Frame f, int addr, int n, int page, bool set)
        {
            byte[] b = f.Bytes;
            const int p = 11;
            string value;
            switch (addr)
            {
                case 0x01:
                    Explain(Add(f, p, n, "SN 号", "已隐藏", "ASCII 固定长度字符串。", 9, false, true),
                        "SN 是 ASCII 文本：每个字节是一个字符的编码，例如 0x41 是字母 A、0x31 是数字 1。", "这里属于设备身份信息，工具不显示内容。");
                    return "SN 号已隐藏";
                case 0x02:
                    if (n < 6) break;
                    value = Date(b, p);
                    Explain(Add(f, p, 6, "当前时间", value, "年、月、日、时、分、秒各一个字节，年份以 2000 为基准。", 10), DateSteps(b, p));
                    if (n >= 7)
                    {
                        int zone = unchecked((sbyte)b[p + 6]);
                        Explain(Add(f, p + 6, 1, "时区参数", "UTC" + (zone / 2.0).ToString("+0.0;-0.0;+0.0", Culture), "SDK 用有符号数存储半小时偏移。", 10),
                            new[] { ByteAt(p + 6, b[p + 6]) + "。" }.Concat(SignSteps("位置 " + (p + 6) + " 的字节 " + Hex(b[p + 6]), (ulong)b[p + 6], 1, zone)).Concat(new[]
                            { "单位是半小时：" + zone + " ÷ 2 = " + (zone / 2.0).ToString("0.0", Culture) + " 小时 → UTC" + (zone / 2.0).ToString("+0.0;-0.0;+0.0", Culture) + "。" }).ToArray());
                    }
                    if (n >= 8)
                        Explain(Add(f, p + 7, 1, "对时方式", EnumValue(b[p + 7], "未定义", "蓝牙对时", "平台对时"), "参数表长度列写 7，备注定义第 8 字节。01 带时区时间，02 为 UTC 时间。", 10),
                            EnumSteps(p + 7, b[p + 7], "未定义", "蓝牙对时", "平台对时"));
                    Raw(f, p + 8, n - 8, 10);
                    return value;
                case 0x0B:
                    if (n != 4) break;
                    return Scaled(f, p, 4, "最大限制电流", 0.1, "A", 10, true);
                case 0x0D:
                    if (n != 1) break;
                    return AddEnum(f, p, "桩模式", 10, Modes);
                case 0x1F:
                    if (n != 1) break;
                    switch (b[p])
                    {
                        case 0x11: value = "进入蓝牙工作模式"; break;
                        case 0x22: value = "Octopus 模式"; break;
                        case 0x55: value = "进入 ATE 模式"; break;
                        case 0xAA: value = "退出"; break;
                        default: value = Unknown(b[p]); break;
                    }
                    Explain(Add(f, p, 1, "ATE 操作码", value, "0x22 为安装商 App 使用的 Octopus 蓝牙模式。", 11),
                        ByteAt(p, b[p]) + "。", "查操作码表：0x11 进入蓝牙工作模式，0x22 Octopus 模式，0x55 进入 ATE 模式，0xAA 退出 → " + value + "。");
                    return value;
                case 0x42:
                    if (n < 4) break;
                    return LoadBalance(f, p, n);
                case 0x4B:
                    if (n != 1) break;
                    return AddEnum(f, p, "电网类型", 11, "IT", "TT", "TN");
                case 0x53:
                    if (n != 1) break;
                    return AddEnum(f, p, "过压阈值", 11, "253 V", "262.2 V");
                case 0xBF:
                    if (n < 32) break;
                    string key = Ascii(b, p, 32);
                    Explain(Add(f, p, 32, "参数键", key, "参数地址 191 的前 32 字节。", 12), AsciiSteps(b, p, 32));
                    if (key == "LightIntensity")
                    {
                        f.Function = (set ? "设置" : "查询") + "指示灯亮度";
                        f.Api = "ENBLEChargeInfoService." + (set ? "set" : "get") + "LightIntensity";
                        if (!set) f.Decoder = "ENBLEChargeInfoUnPack.parseGetLightIntensityResponse";
                        if (n == 32 || (!set && f.Command == 0xAA)) { Raw(f, p + 32, n - 32, 23); return ""; }
                        value = b[p + 32] == 0 ? "关闭" : b[p + 32] == 15 ? "低" : b[p + 32] == 50 ? "中" : b[p + 32] == 100 ? "高" : "原始亮度 " + b[p + 32];
                        Explain(Add(f, p + 32, 1, "指示灯亮度", value + "（" + b[p + 32] + "）", "安装商 App 枚举 0、15、50、100。通用 SDK 将非内置档位保留为 Custom。", 23),
                            ByteAt(p + 32, b[p + 32]) + "。", "安装商 App 的档位：0 关闭，15 低，50 中，100 高 → " + b[p + 32] + " 是「" + value + "」。");
                        Raw(f, p + 33, n - 33, 23);
                        return value;
                    }
                    bool secret = key == "AuthorizationKey" || key == "chargePointID";
                    Add(f, p + 32, n - 32, "参数值", secret ? "已隐藏" : "字段未覆盖", "参见附录 5.7。", 23, false, secret);
                    return secret ? "参数值已隐藏" : "";
                case 0xC3:
                    if (n < 6) break;
                    return Connected(f, p, n);

        private static string Bits(byte[] b, int start, int count, string group, string[][] names)
        {
            // Every set bit by name and position, e.g. "拆机告警（warnings[0] bit5）".
            var found = new List<string>();
            for (int i = 0; i < count; i++)
                for (int bit = 0; bit < 8; bit++)
                {
                    if ((b[start + i] & (1 << bit)) == 0) continue;
                    string name = i < names.Length ? names[i][bit] : null;
                    found.Add((name ?? "预留位") + "（" + group + "[" + i + "] bit" + bit + "）");
                }
            return found.Count == 0 ? "无置位" : String.Join("、", found);
        }

        private static string Scaled(Frame f, int p, int n, string name, double scale, string unit, int page, bool signed)
        {
            ulong raw = ReadNumber(f.Bytes, p, n);
            long number = (long)raw;
            if (signed && n == 2) number = unchecked((short)raw);
            if (signed && n == 4) number = unchecked((int)raw);
            string format = scale == 1 ? "0" : scale == 0.1 ? "0.0" : "0.00";
            string value = (number * scale).ToString(format, Culture) + (unit.Length > 0 ? " " + unit : "");
            var steps = new List<string>(NumberSteps(f.Bytes, p, n, (long)raw));
            if (signed && n > 1) steps.AddRange(SignSteps("上一步拼成的 0x" + raw.ToString("X" + (n * 2)), raw, n, number));
            if (scale != 1) steps.Add("乘以单位 " + scale.ToString(Culture) + "：" + number + " × " + scale.ToString(Culture) + " = " + value + "。");
            else if (unit.Length > 0) steps.Add("单位是 " + unit + " → " + value + "。");
            Explain(Add(f, p, n, name, value, "小端数值 " + number + " × " + scale.ToString(Culture) + "。", page), steps.ToArray());
            return value;
        }

        private static string BitsOf(int value) { return Convert.ToString(value & 0xFF, 2).PadLeft(8, '0').Insert(4, " "); }

        // A number in binary, in groups of four, e.g. "1111 1111 1110 0110".
        private static string Binary(ulong value, int bits)
        {
            ulong mask = bits >= 64 ? ulong.MaxValue : (1UL << bits) - 1;
            string text = Convert.ToString((long)(value & mask), 2).PadLeft(bits, '0');
            var grouped = new StringBuilder();
            for (int i = 0; i < text.Length; i++) { if (i > 0 && i % 4 == 0) grouped.Append(' '); grouped.Append(text[i]); }
            return grouped.ToString();
        }

        // "0x10 = 十进制 16 = 二进制 0001 0000"
        private static string ByteText(int value) { return Hex(value & 0xFF) + " = 十进制 " + (value & 0xFF) + " = 二进制 " + BitsOf(value); }

        // "位置 7 的字节 0x10 = 十进制 16 = 二进制 0001 0000": every step names the byte it reads.
        private static string ByteAt(int position, int value) { return "位置 " + position + " 的字节 " + ByteText(value); }

        // The bits from..from+count-1 of a byte, written high bit first, e.g. "0001".
        private static string BitText(int value, int from, int count)
        {
            return Convert.ToString((value >> from) & ((1 << count) - 1), 2).PadLeft(count, '0');
        }

        // "0001 0000 的最右边 2 位（bit0–1）= 00": which bits of which byte, counted in plain words.
        private static string BitPart(int value, int from, int count)
        {
            string where = from == 0 ? (count == 1 ? "最右边一位" : "最右边 " + count + " 位")
                : from + count == 8 ? (count == 1 ? "最左边一位" : "最左边 " + count + " 位")
                : count == 1 ? "右数第 " + (from + 1) + " 位" : "右数第 " + (from + 1) + "–" + (from + count) + " 位";
            string label = count == 1 ? "bit" + from : "bit" + from + "–" + (from + count - 1);
            return BitsOf(value) + " 的" + where + "（" + label + "）= " + BitText(value, from, count);
        }

        private static string[] SignSteps(string what, ulong raw, int bytes, long value)
        {
            // Signed numbers in plain words: the leftmost bit of the number is its sign; a negative number is
            // "flip every bit, add 1". "what" names the number, e.g. "上一步拼成的 0xFFFFFFE6".
            int bits = bytes * 8;
            ulong top = 1UL << (bits - 1);
            if ((raw & top) == 0)
                return new string[0];   // a positive number reads the same either way, so nothing to explain
            ulong mask = bits == 64 ? ulong.MaxValue : (1UL << bits) - 1;
            ulong flipped = ~raw & mask;
            string digits = "X" + (bytes * 2);
            string total = ((decimal)mask + 1).ToString(Culture);
            return new[]
            {
                "这个字段可能是负数（比如电流的方向），所以按「有符号数」读：把" + what + " 写成 " + bits + " 位二进制是 " + Binary(raw, bits)
                    + "，它最左边那一位（bit" + (bits - 1) + "）是正负号，0 是正数，1 是负数；这里是 1 → 负数。",
                "    负数怎么算——「取反加一」：",
                "    ① 把 " + Binary(raw, bits) + " 的每一位 0 变 1、1 变 0 → " + Binary(flipped, bits) + "，也就是 0x" + flipped.ToString(digits, Culture) + " = " + flipped.ToString(Culture) + "；",
                "    ② 再加 1：" + flipped.ToString(Culture) + " + 1 = " + (flipped + 1).ToString(Culture) + "；",
                "    ③ 前面加负号 → " + value.ToString(Culture) + "。",
                "    验算：" + bits + " 位一共能表示 2^" + bits + " = " + total + " 个数，负数是从这个总数倒着数回来的（" + bits + " 位全是 1 的 0x"
                    + mask.ToString(digits, Culture) + " 就是 -1）：0x" + raw.ToString(digits, Culture) + " = " + raw.ToString(Culture) + "，" + raw.ToString(Culture) + " − " + total + " = " + value.ToString(Culture) + "。"
            };
        }

        private static string[] NumberSteps(byte[] b, int p, int n, long value)
        {
            // Little endian: the byte at the lowest position is the lowest part, so the number is read backwards.
            if (n == 1) return new[] { ByteAt(p, b[p]) + "。" };
            var reversed = new StringBuilder();
            for (int i = n - 1; i >= 0; i--) reversed.Append(b[p + i].ToString("X2"));
            return new[]
            {
                "位置 " + p + "–" + (p + n - 1) + " 这 " + n + " 个字节是 " + BitConverter.ToString(b, p, n).Replace('-', ' ') + "，按小端存储：位置 " + p + " 的 "
                    + b[p].ToString("X2") + " 是最低位字节，位置 " + (p + n - 1) + " 的 " + b[p + n - 1].ToString("X2") + " 是最高位字节。从位置 "
                    + (p + n - 1) + " 往回读到位置 " + p + "，拼成 0x" + reversed + "。",
                "0x" + reversed + " = 十进制 " + value.ToString(Culture) + "。"
            };
        }

        private static string[] EnumSteps(int position, int value, params string[] options)
        {
            var mapping = new string[options.Length];
            for (int i = 0; i < options.Length; i++) mapping[i] = i + " " + options[i];
            return new[] { ByteAt(position, value) + "。", "查表：" + String.Join("，", mapping) + " → " + value + " 是「" + EnumValue(value, options) + "」。" };
        }

        private static string[] BitSteps(byte[] b, int start, int count, string group, string[][] names)
        {
            // Each byte in binary, then every bit that is 1 and what it means.
            int end = start + count - 1;
            var steps = new List<string> { "位置 " + start + "–" + end + " 这 " + count + " 个字节就是 " + group + "[0]–[" + (count - 1) + "]，每个字节 8 位，每一位代表一项，是 1 就表示那一项正在发生。" };
            bool any = false;
            for (int i = 0; i < count; i++)
            {
                int value = b[start + i];
                if (value == 0) continue;
                any = true;
                steps.Add(group + "[" + i + "] 是" + ByteAt(start + i, value) + "：");
                for (int bit = 0; bit < 8; bit++)
                {
                    if ((value & (1 << bit)) == 0) continue;
                    string name = i < names.Length ? names[i][bit] : null;
                    steps.Add("    " + BitPart(value, bit, 1) + " → " + (name ?? "预留位（协议没有定义）") + "。");
                }
            }
            if (!any) steps.Add("位置 " + start + "–" + end + " 的 " + count + " 个字节全是 0x00 → 没有任何一位是 1 → 无置位。");
            else if (count > 1) steps.Add("其余字节都是 0x00，没有置位。");
            return steps.ToArray();
        }

        private static string[] DateSteps(byte[] b, int p)
        {
            if (ReadNumber(b, p, 6) == 0) return new[] { "位置 " + p + "–" + (p + 5) + " 的 6 个字节全是 0x00 → 没有记录这个时间。" };
            string[] parts = { "年", "月", "日", "时", "分", "秒" };
            var steps = new List<string> { "位置 " + p + "–" + (p + 5) + " 的 6 个字节依次是年、月、日、时、分、秒，每个字节直接换成十进制：" };
            for (int i = 0; i < 6; i++)
                steps.Add("    " + parts[i] + "（位置 " + (p + i) + "）：" + Hex(b[p + i]) + " = " + b[p + i] + (i == 0 ? " → 2000 + " + b[p] + " = " + (2000 + b[p]) + " 年" : ""));
            steps.Add("合起来 → " + Date(b, p) + "。");
            return steps.ToArray();
        }

        private static string[] AsciiSteps(byte[] b, int p, int n)
        {
            // ASCII text: one byte per character, 0x00 ends the text.
            var steps = new List<string> { "位置 " + p + "–" + (p + n - 1) + " 这 " + n + " 个字节是 ASCII 文本：每个字节是一个字符的编码，读到 0x00 就结束，后面的 0x00 是补位。" };
            var chars = new List<string>();
            int i = 0;
            for (; i < n && b[p + i] != 0 && chars.Count < 16; i++)
                chars.Add(b[p + i].ToString("X2") + "=" + (b[p + i] >= 0x20 && b[p + i] < 0x7F ? "'" + (char)b[p + i] + "'" : "?"));
            if (chars.Count == 0) steps.Add("位置 " + p + " 的第一个字节就是 0x00 → 空字符串。");
            else
            {
                steps.Add("从位置 " + p + " 开始逐个对照：" + String.Join("  ", chars) + (i < n && b[p + i] != 0 ? " …" : "") + "。");
                steps.Add("连起来 → \"" + Ascii(b, p, n) + "\"。");
            }
            return steps.ToArray();
        }
        private static string AddEnum(Frame f, int p, string name, int page, params string[] options)
        {
            string value = EnumValue(f.Bytes[p], options);
            var mapping = new string[options.Length];
            for (int i = 0; i < options.Length; i++) mapping[i] = i + " " + options[i];
            Explain(Add(f, p, 1, name, value, "枚举 " + string.Join("，", mapping) + "。", page), EnumSteps(p, f.Bytes[p], options));
            return value;
        }

        private static void Invalid(Frame f, int length, int page, string message)
        {
            Error(f, message);
            Raw(f, 7, length, page);
        }

        private static void Error(Frame f, string message)
        {
            f.Error = f.Error.Length == 0 ? message : f.Error + " " + message;
        }

        private static void Raw(Frame f, int start, int length, int page)
        {
            bool zeros = length > 0 && start >= 0 && start + length <= f.Bytes.Length && f.Bytes.Skip(start).Take(length).All(item => item == 0);
            Explain(Add(f, start, length, "原始数据", length + " 字节", "当前字段表未映射此字节范围，原样保留。", page),
                "协议字段表没有给这 " + length + " 个字节定义含义，工具只原样显示。" + (zeros ? "它们全是 0x00，多半是补位。" : ""));
        }

        private static Field Add(Frame f, int start, int length, string name, string value, string explanation, int page, bool header = false, bool hide = false)
        {
            if (length <= 0 || start < 0 || start + length > f.Bytes.Length) return null;
            string hex = hide ? new string('*', length * 3 - 1).Replace("***", "** ") : BitConverter.ToString(f.Bytes, start, length).Replace('-', ' ');
            var field = new Field { Start = start, End = start + length - 1, Name = name, Hex = hex, Value = value, Explanation = explanation, Page = page, IsHeader = header };
            f.Fields.Add(field);
            return field;
        }

        private static void Explain(Field field, params string[] steps)
        {
            if (field != null && steps != null) field.Detail.AddRange(steps);
        }

        private static ulong ReadNumber(byte[] bytes, int start, int length)
        {
            ulong value = 0;
            for (int i = 0; i < length; i++) value |= (ulong)bytes[start + i] << (8 * i);
            return value;
        }

        private static string Date(byte[] b, int start)
        {
            if (ReadNumber(b, start, 6) == 0) return "未记录（全零）";
            return string.Format(Culture, "{0:D4}-{1:D2}-{2:D2} {3:D2}:{4:D2}:{5:D2}", 2000 + b[start], b[start + 1], b[start + 2], b[start + 3], b[start + 4], b[start + 5]);
        }

        private static string Ascii(byte[] bytes, int start, int length)
        {
            int end = start;
            while (end < start + length && bytes[end] != 0) end++;
            return Encoding.ASCII.GetString(bytes, start, end - start);
        }

        private static string Hex(int value) { return "0x" + value.ToString("X2"); }
        private static string Unknown(int value) { return "未定义（" + Hex(value) + "）"; }
        private static string EnumValue(int value, params string[] options) { return value >= 0 && value < options.Length ? options[value] : Unknown(value); }
    }
