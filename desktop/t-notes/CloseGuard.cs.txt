using System;
using System.Runtime.InteropServices;
using System.Text;
using System.Threading;

namespace Codex.Toolbar
{
    // A click on the close button of the T Notes window minimizes the window instead, so the next click on T shows it
    // at once with everything as it was. With Shift held the click goes through and the window closes. Only left clicks
    // are looked at, on a thread of its own with its own message loop, so a busy toolbar never slows the mouse down.
    public static class CloseGuard
    {
        public static string Title = "T Notes v4";
        public static string ClassName = "Chrome_WidgetWin_1";
        public static int Minimized;

        private delegate IntPtr HookProc(int code, IntPtr wParam, IntPtr lParam);

        [StructLayout(LayoutKind.Sequential)]
        private struct POINT { public int X; public int Y; }

        [StructLayout(LayoutKind.Sequential)]
        private struct MSLLHOOKSTRUCT { public POINT Point; public uint MouseData; public uint Flags; public uint Time; public IntPtr Extra; }

        [StructLayout(LayoutKind.Sequential)]
        private struct RECT { public int Left; public int Top; public int Right; public int Bottom; }

        [StructLayout(LayoutKind.Sequential)]
        private struct MSG { public IntPtr Window; public uint Message; public IntPtr WParam; public IntPtr LParam; public uint Time; public POINT Point; }

        [DllImport("user32.dll", SetLastError = true)] private static extern IntPtr SetWindowsHookEx(int id, HookProc proc, IntPtr module, uint threadId);
        [DllImport("user32.dll")] private static extern bool UnhookWindowsHookEx(IntPtr hook);
        [DllImport("user32.dll")] private static extern IntPtr CallNextHookEx(IntPtr hook, int code, IntPtr wParam, IntPtr lParam);
        [DllImport("user32.dll")] private static extern int GetMessage(out MSG message, IntPtr window, uint first, uint last);
        [DllImport("user32.dll")] private static extern bool PostThreadMessage(uint threadId, uint message, IntPtr wParam, IntPtr lParam);
        [DllImport("kernel32.dll")] private static extern uint GetCurrentThreadId();
        [DllImport("kernel32.dll", CharSet = CharSet.Unicode)] private static extern IntPtr GetModuleHandle(string name);
        [DllImport("user32.dll")] private static extern IntPtr WindowFromPoint(POINT point);
        [DllImport("user32.dll")] private static extern IntPtr GetAncestor(IntPtr window, uint flags);
        [DllImport("user32.dll", CharSet = CharSet.Unicode)] private static extern int InternalGetWindowText(IntPtr window, StringBuilder text, int size);
        [DllImport("user32.dll", CharSet = CharSet.Unicode)] private static extern int GetClassName(IntPtr window, StringBuilder text, int size);
        [DllImport("user32.dll")] private static extern bool GetWindowRect(IntPtr window, out RECT rect);
        [DllImport("dwmapi.dll")] private static extern int DwmGetWindowAttribute(IntPtr window, int attribute, out RECT rect, int size);
        [DllImport("user32.dll")] private static extern uint GetDpiForWindow(IntPtr window);
        [DllImport("user32.dll")] private static extern short GetAsyncKeyState(int key);
        [DllImport("user32.dll")] private static extern bool ShowWindowAsync(IntPtr window, int command);
        [DllImport("user32.dll")] private static extern IntPtr SetThreadDpiAwarenessContext(IntPtr context);

        private static readonly HookProc Proc = OnMouse;
        private static IntPtr hook;
        private static uint threadId;
        private static bool swallowRelease;

        public static bool Running { get { return threadId != 0 && hook != IntPtr.Zero; } }

        public static void Start()
        {
            if (threadId != 0)
            {
                return;
            }
            var ready = new ManualResetEvent(false);
            var thread = new Thread(() => Run(ready));
            thread.IsBackground = true;
            thread.Name = "TNotesCloseGuard";
            thread.Start();
            ready.WaitOne(3000);
        }

        public static void Stop()
        {
            if (threadId != 0)
            {
                // WM_QUIT ends the message loop, which removes the hook.
                PostThreadMessage(threadId, 0x0012, IntPtr.Zero, IntPtr.Zero);
            }
        }

        private static void Run(ManualResetEvent ready)
        {
            // Screen positions in physical pixels, as the mouse hook reports them.
            try { SetThreadDpiAwarenessContext(new IntPtr(-4)); } catch (EntryPointNotFoundException) { }
            threadId = GetCurrentThreadId();
            hook = SetWindowsHookEx(14, Proc, GetModuleHandle(null), 0);
            ready.Set();
            MSG message;
            while (GetMessage(out message, IntPtr.Zero, 0, 0) > 0)
            {
            }
            UnhookWindowsHookEx(hook);
            hook = IntPtr.Zero;
            threadId = 0;
        }

        private static IntPtr OnMouse(int code, IntPtr wParam, IntPtr lParam)
        {
            if (code >= 0)
            {
                int message = wParam.ToInt32();
                if (message == 0x0201)
                {
                    var info = (MSLLHOOKSTRUCT)Marshal.PtrToStructure(lParam, typeof(MSLLHOOKSTRUCT));
                    IntPtr window = CloseButtonAt(info.Point.X, info.Point.Y);
                    if (window != IntPtr.Zero && (GetAsyncKeyState(0x10) & 0x8000) == 0)
                    {
                        // The press and its release never reach the window, so its button does not close it.
                        swallowRelease = true;
                        ShowWindowAsync(window, 6);
                        Minimized++;
                        return new IntPtr(1);
                    }
                }
                else if (message == 0x0202 && swallowRelease)
                {
                    swallowRelease = false;
                    return new IntPtr(1);
                }
            }
            return CallNextHookEx(hook, code, wParam, lParam);
        }

        // The T Notes window whose close button is at this screen point, or zero. The button is the top right
        // 46 x 32 pixels (at 100% scaling) of the window's visible frame, as Edge draws it.
        public static IntPtr CloseButtonAt(int x, int y)
        {
            var point = new POINT();
            point.X = x;
            point.Y = y;
            IntPtr window = GetAncestor(WindowFromPoint(point), 2);
            if (window == IntPtr.Zero)
            {
                return IntPtr.Zero;
            }
            var text = new StringBuilder(64);
            InternalGetWindowText(window, text, text.Capacity);
            if (text.ToString() != Title)
            {
                return IntPtr.Zero;
            }
            var name = new StringBuilder(64);
            GetClassName(window, name, name.Capacity);
            if (name.ToString() != ClassName)
            {
                return IntPtr.Zero;
            }
            RECT frame;
            if (DwmGetWindowAttribute(window, 9, out frame, Marshal.SizeOf(typeof(RECT))) != 0 && !GetWindowRect(window, out frame))
            {
                return IntPtr.Zero;
            }
            uint dpi = GetDpiForWindow(window);
            double scale = dpi > 0 ? dpi / 96.0 : 1.0;
            int width = (int)Math.Round(46 * scale);
            int height = (int)Math.Round(32 * scale);
            bool inside = x >= frame.Right - width && x < frame.Right && y >= frame.Top && y < frame.Top + height;
            return inside ? window : IntPtr.Zero;
        }
    }
}
